2. Теоретические сведения: Работа со списками ListView

Класс javafx.scene.control.ListView отображает прокручиваемый список элементов. В приложении список типизируется строками — ListView.

Компонент связывается с коллекцией-наблюдателем — ObservableList:

ObservableList items = FXCollections.observableArrayList();
taskListView.setItems(items);
3. Исходный код контроллера списка (src/com/example/PrimaryController.java)
package com.example;

import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.event.ActionEvent;
import javafx.fxml.FXML;
import javafx.scene.control.Button;
import javafx.scene.control.ListView;
import javafx.scene.control.TextField;

public class PrimaryController {

    @FXML private TextField taskInput;
    @FXML private ListView taskListView;
    @FXML private Button addButton;
    @FXML private Button deleteButton;
    @FXML private Button clearButton;

    private ObservableList tasks = FXCollections.observableArrayList();

    @FXML
    public void initialize() {
        taskListView.setItems(tasks);
    }

    @FXML
    void onAddClick(ActionEvent event) {
        String text = taskInput.getText().trim();
        if (!text.isEmpty()) {
            tasks.add(text);
            taskInput.clear();
        }
    }

    @FXML
    void onDeleteClick(ActionEvent event) {
        String selected = taskListView.getSelectionModel().getSelectedItem();
        if (selected != null) {
            tasks.remove(selected);
        }
    }

    @FXML
    void onClearClick(ActionEvent event) {
        tasks.clear();
    }
}

4. Исходный код контроллера (src/com/example/MainController.java)
package com.example;

import javafx.event.ActionEvent;
import javafx.fxml.FXML;
import javafx.scene.control.Alert;
import javafx.scene.control.Button;

public class MainController {

    // 9 кнопок игрового поля FXML
    @FXML private Button btn1, btn2, btn3;
    @FXML private Button btn4, btn5, btn6;
    @FXML private Button btn7, btn8, btn9;

    // Текущий символ (первым ходит "X")
    private String symbol = "X";

    @FXML
    void btnClick(ActionEvent event) {
        Button btn = (Button) event.getSource();

        // Если клетка уже занята — выходим
        if (!btn.getText().isEmpty()) {
            return;
        }

        // Ставим символ и применяем цветной CSS-класс
        btn.setText(symbol);
        if (symbol.equals("X")) {
            btn.getStyleClass().add("symbol-x");
        } else {
            btn.getStyleClass().add("symbol-o");
        }

        // Проверяем победу
        if (checkWin()) {
            Alert alert = new Alert(Alert.AlertType.INFORMATION);
            alert.setTitle("Победа!");
            alert.setHeaderText(null);
            alert.setContentText("Победил игрок: " + symbol + "!");
            alert.showAndWait();

            resetGame();
            return;
        }

        // Передаем ход
        if (symbol.equals("X")) {
            symbol = "O";
        } else {
            symbol = "X";
        }
    }

    // Проверка 8 победных линий
    private boolean checkWin() {
        // Горизонтали
        if (checkLine(btn1, btn2, btn3)) return true;
        if (checkLine(btn4, btn5, btn6)) return true;
        if (checkLine(btn7, btn8, btn9)) return true;

        // Вертикали
        if (checkLine(btn1, btn4, btn7)) return true;
        if (checkLine(btn2, btn5, btn8)) return true;
        if (checkLine(btn3, btn6, btn9)) return true;

        // Диагонали
        if (checkLine(btn1, btn5, btn9)) return true;
        if (checkLine(btn3, btn5, btn7)) return true;

        return false;
    }

    // Линия совпала, если все 3 кнопки содержат текущий символ
    private boolean checkLine(Button b1, Button b2, Button b3) {
        boolean isWin = b1.getText().equals(symbol) 
                     && b2.getText().equals(symbol) 
                     && b3.getText().equals(symbol);

        if (isWin) {
            b1.getStyleClass().add("winner-line");
            b2.getStyleClass().add("winner-line");
            b3.getStyleClass().add("winner-line");
        }
        return isWin;
    }

    // Сброс поля после раунда
    private void resetGame() {
        clearButton(btn1); clearButton(btn2); clearButton(btn3);
        clearButton(btn4); clearButton(btn5); clearButton(btn6);
        clearButton(btn7); clearButton(btn8); clearButton(btn9);
        symbol = "X";
    }

    private void clearButton(Button btn) {
        btn.setText("");
        btn.getStyleClass().removeAll("symbol-x", "symbol-o", "winner-line");
    }
}

4. Запуск без рамок и непрерывный ввод с клавиатуры (src/com/example/App.java)

В играх критически важно использовать флаги состояния нажатых клавиш (left, right, isJump, isPause) вместо одиночных событий, чтобы избежать системной задержки автоповтора ОС (Key Repeat Delay):

  • StageStyle.UNDECORATED: Убирает системную рамку окна Windows.
  • setOnKeyPressed: Устанавливает флаг клавиши в true при зажатии.
  • setOnKeyReleased: Сбрасывает флаг в false при отпускании, а по клавише ESC инвертирует режим паузы.

Исходный код главного класса приложения (src/com/example/App.java):

package com.example;

import javafx.application.Application;
import javafx.fxml.FXMLLoader;
import javafx.scene.Scene;
import javafx.scene.input.KeyCode;
import javafx.stage.Stage;
import javafx.stage.StageStyle;
import java.io.IOException;

public class App extends Application {
    @Override
    public void start(Stage stage) throws IOException {
        FXMLLoader fxmlLoader = new FXMLLoader(App.class.getResource("MainView.fxml"));
        Scene scene = new Scene(fxmlLoader.load(), 712, 400);

        // Обработка зажатия клавиш (Управление)
        scene.setOnKeyPressed(event -> {
            if (event.getCode() == KeyCode.A || event.getCode() == KeyCode.LEFT) {
                MainController.left = true;
            }
            if (event.getCode() == KeyCode.D || event.getCode() == KeyCode.RIGHT) {
                MainController.right = true;
            }
            if ((event.getCode() == KeyCode.SPACE || event.getCode() == KeyCode.W || event.getCode() == KeyCode.UP) && !MainController.isJump) {
                MainController.isJump = true;
            }
        });

        // Обработка отпускания клавиш
        scene.setOnKeyReleased(event -> {
            if (event.getCode() == KeyCode.A || event.getCode() == KeyCode.LEFT) {
                MainController.left = false;
            }
            if (event.getCode() == KeyCode.D || event.getCode() == KeyCode.RIGHT) {
                MainController.right = false;
            }
            if (event.getCode() == KeyCode.ESCAPE) {
                MainController.isPause = !MainController.isPause;
            }
        });

        stage.initStyle(StageStyle.UNDECORATED);
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch();
    }
}
5. Игровой контроллер: Анимации, физика и коллизии (src/com/example/MainController.java)

Контроллер объединяет все ключевые игровые системы:

  • Бесконечный скроллинг фона: Два TranslateTransition смещают оба фона на -712 px за 5 секунд с Interpolator.LINEAR, объединенные в ParallelTransition с бесконечным циклом Animation.INDEFINITE.
  • Движение препятствия: enemyTransition циклически перемещает врага справа налево за 3.5 секунды.
  • Игровой цикл AnimationTimer (60 FPS):
    • Горизонтальное перемещение с ограничением по экрану (28 < X < 100);
    • Двухфазный прыжок: подъем до высоты Y = 90, затем гравитационный спуск до земли Y = 181;
    • Проверка коллизии: player.getBoundsInParent().intersects(enemy.getBoundsInParent()) — при пересечении игра останавливается и появляется надпись поражения.

Полный исходный код игрового контроллера (src/com/example/MainController.java):

package com.example;

import javafx.animation.Animation;
import javafx.animation.AnimationTimer;
import javafx.animation.Interpolator;
import javafx.animation.ParallelTransition;
import javafx.animation.TranslateTransition;
import javafx.fxml.FXML;
import javafx.scene.control.Label;
import javafx.scene.image.ImageView;
import javafx.util.Duration;

public class MainController {

    @FXML private ImageView bg1, bg2, player, enemy;
    @FXML private Label labelPause, labelLose;

    // Статические флаги управления с клавиатуры
    public static boolean right = false;
    public static boolean left = false;
    public static boolean isJump = false;
    public static boolean isPause = false;

    // Скорости движения и гравитации
    private int playerSpeed = 3;
    private int jumpDownSpeed = 5;
    private final int BG_WIDTH = 712;

    private ParallelTransition parallelTransition;
    private TranslateTransition enemyTransition;

    // Главный игровой цикл (60 раз в секунду)
    private AnimationTimer timer = new AnimationTimer() {
        @Override
        public void handle(long l) {
            // 1. Обработка состояния паузы
            if (isPause && !labelPause.isVisible()) {
                playerSpeed = 0;
                jumpDownSpeed = 0;
                parallelTransition.pause();
                enemyTransition.pause();
                labelPause.setVisible(true);
            } else if (!isPause && labelPause.isVisible()) {
                labelPause.setVisible(false);
                playerSpeed = 3;
                jumpDownSpeed = 5;
                parallelTransition.play();
                enemyTransition.play();
            }

            // 2. Горизонтальное движение персонажа в границах экрана
            if (right && player.getLayoutX() < 100) {
                player.setLayoutX(player.getLayoutX() + playerSpeed);
            }
            if (left && player.getLayoutX() > 28) {
                player.setLayoutX(player.getLayoutX() - playerSpeed);
            }

            // 3. Физика прыжка и гравитации
            if (isJump && player.getLayoutY() > 90) {
                player.setLayoutY(player.getLayoutY() - playerSpeed); // Фаза взлета
            } else if (player.getLayoutY() <= 181) {
                isJump = false; // Достигли пика прыжка
                if (player.getLayoutY() < 181) {
                    player.setLayoutY(player.getLayoutY() + jumpDownSpeed); // Падение вниз (гравитация)
                }
            }

            // 4. Детекция столкновений (Коллизии хитбоксов)
            if (player.getBoundsInParent().intersects(enemy.getBoundsInParent())) {
                labelLose.setVisible(true);
                playerSpeed = 0;
                jumpDownSpeed = 0;
                parallelTransition.pause();
                enemyTransition.pause();
            }
        }
    };

    @FXML
    void initialize() {
        // 1. Запуск бесшовной циклической прокрутки фона
        TranslateTransition bgOneTransition = new TranslateTransition(Duration.millis(5000), bg1);
        bgOneTransition.setFromX(0);
        bgOneTransition.setToX(-BG_WIDTH);
        bgOneTransition.setInterpolator(Interpolator.LINEAR);

        TranslateTransition bgTwoTransition = new TranslateTransition(Duration.millis(5000), bg2);
        bgTwoTransition.setFromX(0);
        bgTwoTransition.setToX(-BG_WIDTH);
        bgTwoTransition.setInterpolator(Interpolator.LINEAR);

        parallelTransition = new ParallelTransition(bgOneTransition, bgTwoTransition);
        parallelTransition.setCycleCount(Animation.INDEFINITE);
        parallelTransition.play();

        // 2. Запуск движения препятствия справа налево
        enemyTransition = new TranslateTransition(Duration.millis(3500), enemy);
        enemyTransition.setFromX(0);
        enemyTransition.setToX(-BG_WIDTH - 100);
        enemyTransition.setInterpolator(Interpolator.LINEAR);
        enemyTransition.setCycleCount(Animation.INDEFINITE);
        enemyTransition.play();

        // 3. Старт игрового цикла
        timer.start();
    }
}

package com.example.model;

import java.util.ArrayList;
import java.util.List;

public class Notebook {
    private final List notes = new ArrayList<>();

    public void addNote(Note note) {
        notes.add(note);
    }

    public List getNotes() {
        return notes;
    }

    public void editNoteAt(int index, Note note) {
        if (index >= 0 && index < notes.size()) {
            notes.set(index, note);
        }
    }
}
